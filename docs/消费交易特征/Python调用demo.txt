#coding=utf-8
import time
import random
import string
import hashlib
import json
import requests

def generate_nonce(sceneid):
    """生成nonce：sceneid加8位随机字符串"""
    random_str = ''.join(random.choices(string.ascii_letters + string.digits, k=8))
    return sceneid + random_str

def generate_sign(params, appkey):
    """
    生成签名
    1. 参数过滤（排除空值）
    2. ASCII排序
    3. 拼接键值对（key=value形式）
    4. 附加appkey
    5. MD5加密
    """
    # 过滤空值参数
    filtered_params = {k: v for k, v in params.items() if v is not None and v != ""}
    
    # 按ASCII码升序排列参数
    sorted_params = sorted(filtered_params.items(), key=lambda x: x)
    
    # 拼接键值对（key=value&key=value形式）
    sign_string = "&".join([f"{k}={v}" for k, v in sorted_params])
    
    # 附加appkey得到待签名字符串
    sign_string += f"&appkey={appkey}"

    print("sign_string",sign_string)
    
    # MD5加密（返回小写）
    sign = hashlib.md5(sign_string.encode('utf-8')).hexdigest()
    
    return sign

def make_request():
    # 固定URL（不参与签名）
    url = "https://api.data-bean.cn"
    
    # 固定Content-Type
    headers = {
        "Content-Type": "application/json"
    }
    
    # 获取当前时间戳（13位）
    reqtime = str(int(time.time() * 1000))
    
    # 生成nonce
    sceneid = "your_sceneid_here"
    nonce = generate_nonce(sceneid)
    
    # 请求参数
    params = {
        #"reqbizno": "15",
        "name": "张三",
        #"mobile": "15825521414",
        #"reasoncode": "01",
        "idcard": "43032120030303332",
        #"authletdate": "2025-7-10",
        # "bankcard": "",
        #"photo": "",
        #"authlet": "abc123"
    }
    
    # 假设的appkey（实际使用时替换为真实的appkey）
    appkey = "your_appkey_here"  # 请替换为实际的appkey
    
    # 构建请求数据（注意：params直接作为字典传递，不转为JSON字符串）
    request_data = {
        "procode": "verifyidcard",
        "sceneid": sceneid,
        "reqtime": reqtime,
        "nonce": nonce,
        "sign": "",  # 先留空，后面计算
        "params": params
    }
    
    # 计算sign（只对procode, sceneid, reqtime, nonce, params的键值对签名）
    # 注意：根据描述，params的值不转为JSON字符串，直接参与签名
    # 所以我们需要将params的键值对提取出来参与签名
    sign_params = {
        "procode": request_data["procode"],
        "sceneid": request_data["sceneid"],
        "reqtime": request_data["reqtime"],
        "nonce": request_data["nonce"]
    }
    
    # 将params的键值对也加入签名参数
    for k, v in request_data["params"].items():
        sign_params[k] = v
    
    # 计算sign
    sign = generate_sign(sign_params, appkey)
    request_data["sign"] = sign
    
    # 发送POST请求（注意：发送时params仍然是嵌套结构）
    # 但签名计算时params是扁平化的

    response = requests.post(
        url,
        headers=headers,
        data=json.dumps({
            "procode": request_data["procode"],
            "sceneid": request_data["sceneid"],
            "reqtime": request_data["reqtime"],
            "nonce": request_data["nonce"],
            "sign": request_data["sign"],
            "params": request_data["params"]
        }, ensure_ascii=False)
    )
    
    duration = response.elapsed.total_seconds()* 1000
    print(">>>---START--->>>")
    # 打印响应
    print("------")
    print("Response Content:", response.text)
    #print("param:",params)
    print(">>>---END--->>>")
    # print("sign",sign)


if __name__ == "__main__":
    make_request()
